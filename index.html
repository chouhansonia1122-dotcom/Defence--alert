async function getFeed(category, query) { const feedUrl = "https://news.google.com/rss/search?q=" + encodeURIComponent(query) + "&hl=en-IN&gl=IN&ceid=IN:en"; const response = await fetch( "https://proxy.cors.dev/" + feedUrl, { credentials: "omit" } ); if (!response.ok) { throw new Error("News feed unavailable: " + response.status); } const xmlText = await response.text(); const xml = new DOMParser().parseFromString( xmlText, "application/xml" ); if (xml.querySelector("parsererror")) { throw new Error("News feed format error"); } return [...xml.querySelectorAll("item")].map(item => { const read = tag => item.querySelector(tag)?.textContent.trim() || ""; return { title: read("title"), description: cleanText(read("description")).slice(0, 900), link: read("link"), published: read("pubDate"), category }; }); }
